// ── Data unchanged ──────────────────────────────────────────
function generateCandles(base, days = 300, volatility = 0.015) {
  const candles = [];
  let price = base;
  const now = Date.now();
  for (let i = days; i >= 0; i--) {
    const d = new Date(now - i * 24 * 3600 * 1000);
    if (d.getDay() === 0 || d.getDay() === 6) continue;
    const change = (Math.random() - 0.48) * volatility;
    const open = price;
    const close = price * (1 + change);
    const high = Math.max(open, close) * (1 + Math.random() * 0.008);
    const low = Math.min(open, close) * (1 - Math.random() * 0.008);
    candles.push({ t: d, o: +open.toFixed(0), h: +high.toFixed(0), l: +low.toFixed(0), c: +close.toFixed(0), v: Math.round(500 + Math.random() * 4500) });
    price = close;
  }
  return candles;
}

// ── Indicators ──────────────────────────────────────────────
function calcSMA(data, period) {
  return data.map((_, i) =>
    i < period - 1 ? null : data.slice(i - period + 1, i + 1).reduce((s, c) => s + c.c, 0) / period
  );
}
function calcEMA(data, period) {
  const k = 2 / (period + 1);
  const ema = [];
  let prev = null;
  data.forEach((d, i) => {
    if (i < period - 1) { ema.push(null); return; }
    if (prev === null) prev = data.slice(0, period).reduce((s, c) => s + c.c, 0) / period;
    prev = d.c * k + prev * (1 - k);
    ema.push(prev);
  });
  return ema;
}
function calcRSI(data, period = 14) {
  return data.map((_, i) => {
    if (i < period) return null;
    let gains = 0, losses = 0;
    for (let j = i - period + 1; j <= i; j++) {
      const d = data[j].c - data[j - 1].c;
      if (d > 0) gains += d; else losses -= d;
    }
    const rs = gains / (losses || 0.001);
    return 100 - 100 / (1 + rs);
  });
}

// ── Chart rendering helpers ─────────────────────────────────
// Extracts pure drawing logic — no React state dependencies
function buildScales({ data, W, H, mainH, PAD_L, PAD_R, PAD_T, PAD_B }) {
  const n = data.length;
  const lows  = data.map(d => d.l);
  const highs = data.map(d => d.h);
  let minP = Math.min(...lows);
  let maxP = Math.max(...highs);
  const pRange = maxP - minP;
  minP -= pRange * 0.05;
  maxP += pRange * 0.12;
  const chartW = W - PAD_L - PAD_R;
  const chartH = mainH - PAD_T - PAD_B;
  const xScale = (i) => PAD_L + (i / (n - 1)) * chartW;
  const yScale = (p) => PAD_T + chartH - ((p - minP) / (maxP - minP)) * chartH;
  const barW   = Math.max(1, (chartW / n) * 0.7);
  return { n, minP, maxP, chartW, chartH, xScale, yScale, barW };
}

  // ── WHEEL ZOOM ────────────────────────────────────────────
  const handleWheel = useCallback((e) => {
    e.preventDefault();
    const delta = e.deltaY > 0 ? 0.05 : -0.05;
    setZoom(z => {
      const range    = z.end - z.start;
      const newRange = Math.max(0.05, Math.min(1, range + delta));
      const center   = (z.start + z.end) / 2;
      return { start: Math.max(0, center - newRange / 2), end: Math.min(1, center + newRange / 2) };
    });
  }, []);

  // currentD — used in JSX for live price display in topbar
  const currentD = (crosshairRef.current && visibleData[crosshairRef.current.idx]) || last;

  // ── Shorthand style aliases ───────────────────────────────
  const { bg, bg2, bg3, bg4, border, text, text2, text3 } = th;

  // ════════════════════════════════════════════════════════
  // JSX — layout unchanged, only performance fixes applied
  // ════════════════════════════════════════════════════════
  return (
    <div style={{ fontFamily: "'JetBrains Mono','Fira Code',monospace", background: bg, color: text, height: "100vh", display: "flex", flexDirection: "column", overflow: "hidden" }}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=JetBrains+Mono:wght@300;400;500;600;700&display=swap');
        ::-webkit-scrollbar{width:4px;height:4px}
        ::-webkit-scrollbar-track{background:transparent}
        ::-webkit-scrollbar-thumb{background:${border};border-radius:2px}
        *{box-sizing:border-box}
        .btn-icon{background:none;border:none;cursor:pointer;padding:4px 8px;border-radius:4px;transition:background .15s;display:flex;align-items:center;gap:4px}
        .btn-icon:hover{background:${bg4}}
        .tab-btn{padding:3px 10px;border-radius:3px;border:none;cursor:pointer;font-size:11px;font-family:inherit;font-weight:600;transition:all .15s}
        input{font-family:inherit}
        .pill{padding:2px 8px;border-radius:20px;font-size:10px;font-weight:700}
        @keyframes fadeIn{from{opacity:0;transform:translateY(-4px)}to{opacity:1;transform:none}}
        .fade-in{animation:fadeIn .2s ease}
        @keyframes pulse{0%,100%{opacity:1}50%{opacity:.4}}
        .live-dot{animation:pulse 2s infinite}
      `}</style>

      {/* TOP BAR */}
      <div style={{ height: 48, background: bg2, borderBottom: `1px solid ${border}`, display: "flex", alignItems: "center", paddingInline: 12, gap: 8, flexShrink: 0, zIndex: 100 }}>
        <div style={{ display: "flex", alignItems: "center", gap: 6, marginRight: 8 }}>
          <div style={{ width: 28, height: 28, background: "linear-gradient(135deg,#00e5a0,#4f8ef7)", borderRadius: 7, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 14, fontWeight: 900 }}>B</div>
          <span style={{ fontSize: 12, fontWeight: 700, color: text }}>BONJOUR<span style={{ color: "#00e5a0" }}>.chart</span></span>
        </div>

        {/* Symbol picker */}
        <div style={{ position: "relative" }}>
          <button onClick={() => setShowSearch(s => !s)}
            style={{ background: bg3, border: `1px solid ${border}`, borderRadius: 6, padding: "5px 12px", color: text, cursor: "pointer", fontSize: 12, fontWeight: 700, display: "flex", alignItems: "center", gap: 8, fontFamily: "inherit" }}>
            <span style={{ fontSize: 14 }}>{info.country}</span>
            <span>{activeSym}</span>
            <span style={{ color: text3, fontWeight: 400 }}>— {info.name}</span>
            <span style={{ color: text3, fontSize: 10 }}>▾</span>
          </button>
          {showSearch && (
            <div className="fade-in" style={{ position: "absolute", top: "calc(100% + 6px)", left: 0, width: 280, background: bg2, border: `1px solid ${border}`, borderRadius: 8, zIndex: 999, boxShadow: "0 12px 40px rgba(0,0,0,.5)", overflow: "hidden" }}>
              <input value={searchQ} onChange={e => setSearchQ(e.target.value)} placeholder="Rechercher un titre BRVM…"
                style={{ width: "100%", padding: "10px 14px", background: bg3, border: "none", borderBottom: `1px solid ${border}`, color: text, fontSize: 12, outline: "none" }} autoFocus />
              <div style={{ maxHeight: 220, overflowY: "auto" }}>
                {filteredSymbols.map(([sym, si]) => {
                  const d = CANDLE_DATA[sym]; const l = d[d.length - 1], p = d[d.length - 2];
                  const chg = ((l.c - p.c) / p.c) * 100;
                  return (
                    <div key={sym} onClick={() => { setActiveSym(sym); setShowSearch(false); setSearchQ(""); }}
                      style={{ display: "flex", alignItems: "center", justifyContent: "space-between", padding: "6px 10px", cursor: "pointer", borderRadius: 4, transition: "background .12s" }}
                      onMouseEnter={e => e.currentTarget.style.background = bg4}
                      onMouseLeave={e => e.currentTarget.style.background = "none"}>
                      <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
                        <span style={{ fontSize: 14 }}>{si.country}</span>
                        <div>
                          <div style={{ fontSize: 11, fontWeight: 700 }}>{sym}</div>
                          <div style={{ fontSize: 10, color: text3 }}>{si.name}</div>
                        </div>
                      </div>
                      <div style={{ textAlign: "right" }}>
                        <div style={{ fontSize: 11, fontWeight: 600 }}>{fmt(l.c)}</div>
                        <div style={{ fontSize: 10, color: chg >= 0 ? th.upC : th.downC, fontWeight: 700 }}>{fmtPct(chg)}</div>
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          )}
        </div>

        {/* Price */}
        <div style={{ display: "flex", alignItems: "center", gap: 8, marginLeft: 4 }}>
          <span style={{ fontSize: 18, fontWeight: 700, color: isUp ? th.upC : th.downC }}>{fmt(currentD.c)}</span>
          <span style={{ fontSize: 12 }}>FCFA</span>
          <span className="pill" style={{ background: isUp ? th.upC + "20" : th.downC + "20", color: isUp ? th.upC : th.downC }}>
            {isUp ? "▲" : "▼"} {fmtPct(changePct)}
          </span>
          <div style={{ display: "flex", gap: 6, fontSize: 10, color: text3 }}>
            <span>H:{fmt(currentD.h)}</span>
            <span>L:{fmt(currentD.l)}</span>
            <span>V:{currentD.v.toLocaleString()}</span>
          </div>
        </div>

        <div style={{ display: "flex", alignItems: "center", gap: 4, marginLeft: 4 }}>
          <div className="live-dot" style={{ width: 6, height: 6, borderRadius: "50%", background: th.upC }} />
          <span style={{ fontSize: 9, color: text3, fontWeight: 600 }}>BRVM LIVE</span>
        </div>
        <div style={{ flex: 1 }} />

        {/* Timeframes */}
        <div style={{ display: "flex", gap: 2, background: bg3, borderRadius: 6, padding: "3px" }}>
          {["1H", "4H", "1J", "1S", "1M", "3M"].map(t => (
            <button key={t} className="tab-btn" onClick={() => setTf(t)}
              style={{ background: tf === t ? "#00e5a0" : "none", color: tf === t ? "#000" : text3 }}>{t}</button>
          ))}
        </div>

        {/* Chart type */}
        <div style={{ display: "flex", gap: 2, background: bg3, borderRadius: 6, padding: "3px", marginLeft: 4 }}>
          {[["candle","🕯"],["bar","▐"],["line","∼"],["area","◭"]].map(([t, ic]) => (
            <button key={t} className="tab-btn" onClick={() => setChartType(t)}
              style={{ background: chartType === t ? bg4 : "none", color: chartType === t ? text : text3, fontSize: 13 }}>{ic}</button>
          ))}
        </div>

        <button className="btn-icon" onClick={() => setIsDark(d => !d)} style={{ color: text2, fontSize: 14 }}>
          {isDark ? "☀️" : "🌙"}
        </button>
      </div>

      {/* MAIN LAYOUT */}
      <div style={{ flex: 1, display: "flex", overflow: "hidden" }}>

        {/* Left toolbar */}
        <div style={{ width: 44, background: bg2, borderRight: `1px solid ${border}`, display: "flex", flexDirection: "column", alignItems: "center", padding: "8px 0", gap: 4, flexShrink: 0 }}>
          {[{ id: null, icon: "↖", tip: "Sélection" }, { id: "trend", icon: "╱", tip: "Tendance" }, { id: "hline", icon: "─", tip: "Horizontal" }, { id: "rect", icon: "▭", tip: "Rectangle" }, { id: "fib", icon: "Φ", tip: "Fibonacci" }]
            .map(({ id, icon, tip }) => (
              <button key={icon} title={tip} className="btn-icon" onClick={() => setDrawMode(d => d === id ? null : id)}
                style={{ width: 32, height: 32, justifyContent: "center", color: drawMode === id ? th.blue : text3, fontSize: 14, background: drawMode === id ? th.blue + "20" : "none" }}>
                {icon}
              </button>
            ))}
          <div style={{ flex: 1 }} />
          <button className="btn-icon" style={{ width: 32, height: 32, justifyContent: "center", color: text3, fontSize: 14 }}>📸</button>
        </div>

        {/* Chart area */}
        <div style={{ flex: 1, display: "flex", flexDirection: "column", overflow: "hidden" }}>

          {/* Indicator pills */}
          <div style={{ height: 34, background: bg2, borderBottom: `1px solid ${border}`, display: "flex", alignItems: "center", paddingInline: 10, gap: 4, flexShrink: 0 }}>
            <span style={{ fontSize: 9, color: text3, fontWeight: 700, textTransform: "uppercase", letterSpacing: ".5px", marginRight: 4 }}>Indicateurs</span>
            {[
              { k: "sma20", label: "SMA 20", color: th.gold },
              { k: "ema50", label: "EMA 50", color: th.purple },
              { k: "volume", label: "Volume", color: th.blue },
              { k: "rsi",   label: "RSI",    color: th.blue },
              { k: "macd",  label: "MACD",   color: th.green },
            ].map(({ k, label, color }) => (
              <button key={k} onClick={() => setIndicators(p => ({ ...p, [k]: !p[k] }))}
                style={{ padding: "2px 8px", borderRadius: 3, border: `1px solid ${indicators[k] ? color : border}`, background: indicators[k] ? color + "20" : "none", color: indicators[k] ? color : text3, cursor: "pointer", fontSize: 10, fontFamily: "inherit", fontWeight: 600, display: "flex", alignItems: "center", gap: 4 }}>
                <span style={{ width: 6, height: 6, borderRadius: "50%", background: indicators[k] ? color : text3 }} />
                {label}
              </button>
            ))}
            <div style={{ flex: 1 }} />
            {indicators.sma20 && <span style={{ fontSize: 10, color: th.gold   }}>SMA20: {fmt(sma20.filter(Boolean).at(-1)   || 0)}</span>}
            {indicators.ema50 && <span style={{ fontSize: 10, color: th.purple, marginLeft: 8 }}>EMA50: {fmt(ema50.filter(Boolean).at(-1) || 0)}</span>}
            {indicators.rsi   && <span style={{ fontSize: 10, color: th.blue,   marginLeft: 8 }}>RSI: {(rsiData.filter(Boolean).at(-1) || 50).toFixed(1)}</span>}
          </div>

          {/* Canvas stack */}
          <div style={{ flex: 1, position: "relative", overflow: "hidden" }}
            onMouseMove={handleMouseMove}
            onMouseLeave={handleMouseLeave}
            onWheel={handleWheel}>
            {/* Layer 1: main chart (expensive, redraws only when data/options change) */}
            <canvas ref={canvasRef} style={{ position: "absolute", inset: 0, width: "100%", height: "100%" }} />
            {/* Layer 2: crosshair (cheap, redraws every RAF tick) */}
            <canvas ref={overlayRef} style={{ position: "absolute", inset: 0, width: "100%", height: "100%", pointerEvents: "none" }} />
          </div>

          {/* Zoom slider */}
          <div style={{ height: 28, background: bg2, borderTop: `1px solid ${border}`, display: "flex", alignItems: "center", paddingInline: 12, gap: 8 }}>
            <span style={{ fontSize: 9, color: text3, fontWeight: 600 }}>ZOOM</span>
            <input type="range" min={5} max={100}
              value={Math.round((1 - (zoom.end - zoom.start)) * 100)}
              onChange={e => {
                const z = 1 - e.target.value / 100;
                const c = (zoom.start + zoom.end) / 2;
                setZoom({ start: Math.max(0, c - z / 2), end: Math.min(1, c + z / 2) });
              }}
              style={{ flex: 1, accentColor: th.blue, height: 4 }} />
            <span style={{ fontSize: 9, color: text3 }}>{visibleData.length} bougies</span>
            <button className="btn-icon" style={{ fontSize: 10, color: text3, padding: "2px 6px" }} onClick={() => setZoom({ start: 0,    end: 1 })}>Tout</button>
            <button className="btn-icon" style={{ fontSize: 10, color: text3, padding: "2px 6px" }} onClick={() => setZoom({ start: 0.85, end: 1 })}>3M</button>
          </div>
        </div>

        {/* Right panel */}
        <div style={{ width: 280, background: bg2, borderLeft: `1px solid ${border}`, display: "flex", flexDirection: "column", flexShrink: 0 }}>
          <div style={{ display: "flex", borderBottom: `1px solid ${border}`, flexShrink: 0 }}>
            {[["watchlist","👁","Liste"], ["order","📋","Ordre"], ["alerts","🔔","Alertes"]].map(([panel, icon, label]) => (
              <button key={panel} onClick={() => setActivePanel(panel)}
                style={{ flex: 1, padding: "10px 4px", background: "none", border: "none", borderBottom: activePanel === panel ? `2px solid ${th.blue}` : "2px solid transparent", color: activePanel === panel ? th.blue : text3, cursor: "pointer", fontSize: 11, fontFamily: "inherit", fontWeight: 600 }}>
                {icon} {label}
              </button>
            ))}
          </div>

          {/* Watchlist — uses React.memo WatchItem */}
          {activePanel === "watchlist" && (
            <div style={{ flex: 1, overflow: "auto" }}>
              <div style={{ padding: "6px 10px 2px", fontSize: 9, color: text3, fontWeight: 700, textTransform: "uppercase", letterSpacing: ".5px" }}>Favoris</div>
              {watchData.filter(d => favorites.includes(d.sym)).map(d => (
                <WatchItem key={d.sym} d={d} isActive={d.sym === activeSym} isUp={d.chg >= 0} onSelect={setActiveSym} th={th} />
              ))}
              <div style={{ padding: "10px 10px 2px", fontSize: 9, color: text3, fontWeight: 700, textTransform: "uppercase", letterSpacing: ".5px" }}>Tous les titres</div>
              {watchData.filter(d => !favorites.includes(d.sym)).map(d => (
                <WatchItem key={d.sym} d={d} isActive={d.sym === activeSym} isUp={d.chg >= 0} onSelect={setActiveSym} th={th} />
              ))}
            </div>
          )}

          {/* Order panel */}
          {activePanel === "order" && (
            <div style={{ flex: 1, padding: 14, display: "flex", flexDirection: "column", gap: 12 }}>
              <div style={{ background: bg3, borderRadius: 8, padding: 12, border: `1px solid ${border}` }}>
                <div style={{ fontSize: 11, fontWeight: 700, marginBottom: 10 }}>{info.country} {activeSym} — {info.name}</div>
                <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 8, fontSize: 10 }}>
                  {[["Dernier", fmt(last.c)+" F"], ["Variation", fmtPct(changePct)], ["Secteur", info.sector], ["Acheteur", fmt(last.c - Math.round(last.c * 0.001))+" F"]].map(([l, v]) => (
                    <div key={l}>
                      <div style={{ color: text3, fontSize: 9 }}>{l}</div>
                      <div style={{ fontWeight: 700, color: l === "Variation" ? (changePct >= 0 ? th.upC : th.downC) : text }}>{v}</div>
                    </div>
                  ))}
                </div>
              </div>
              <div style={{ display: "flex", borderRadius: 6, overflow: "hidden", border: `1px solid ${border}` }}>
                {[["buy","ACHETER",th.upC], ["sell","VENDRE",th.downC]].map(([t, l, c]) => (
                  <button key={t} onClick={() => setOrderTab(t)}
                    style={{ flex: 1, padding: "8px 0", background: orderTab === t ? c + "20" : "none", border: "none", color: orderTab === t ? c : text3, cursor: "pointer", fontSize: 11, fontFamily: "inherit", fontWeight: 700, transition: "all .15s" }}>{l}</button>
                ))}
              </div>
              <div>
                <div style={{ fontSize: 9, color: text3, marginBottom: 4, fontWeight: 600, textTransform: "uppercase" }}>Prix (FCFA)</div>
                <div style={{ display: "flex", gap: 4 }}>
                  <input value={price_} onChange={e => setPrice_(e.target.value)} placeholder={fmt(last.c)}
                    style={{ flex: 1, background: bg3, border: `1px solid ${border}`, borderRadius: 5, padding: "7px 10px", color: text, fontSize: 11, outline: "none" }} />
                  <button onClick={() => setPrice_(String(last.c))} style={{ padding: "0 8px", background: bg3, border: `1px solid ${border}`, borderRadius: 5, color: text3, cursor: "pointer", fontSize: 9, fontFamily: "inherit" }}>Mkt</button>
                </div>
              </div>
              <div>
                <div style={{ fontSize: 9, color: text3, marginBottom: 4, fontWeight: 600, textTransform: "uppercase" }}>Quantité</div>
                <input value={qty} onChange={e => setQty(e.target.value)} placeholder="ex: 10"
                  style={{ width: "100%", background: bg3, border: `1px solid ${border}`, borderRadius: 5, padding: "7px 10px", color: text, fontSize: 11, outline: "none" }} />
              </div>
              {qty && price_ && (
                <div style={{ background: bg3, borderRadius: 6, padding: 10, fontSize: 10 }}>
                  <div style={{ display: "flex", justifyContent: "space-between", color: text3 }}><span>Total</span><span style={{ color: text, fontWeight: 700 }}>{fmt(+qty * +price_)} FCFA</span></div>
                  <div style={{ display: "flex", justifyContent: "space-between", color: text3, marginTop: 4 }}><span>Courtage ~0.6%</span><span>{fmt(+qty * +price_ * 0.006)} FCFA</span></div>
                </div>
              )}
              <button style={{ padding: "10px 0", borderRadius: 6, border: "none", background: orderTab === "buy" ? th.upC : th.downC, color: "#000", fontFamily: "inherit", fontSize: 12, fontWeight: 800, cursor: "pointer" }}>
                {orderTab === "buy" ? "PASSER ORDRE ACHAT" : "PASSER ORDRE VENTE"}
              </button>
              <div style={{ fontSize: 9, color: text3, textAlign: "center", lineHeight: 1.5 }}>Ordre informatif — passez par votre SGI agréée CREPMF</div>
            </div>
          )}

          {/* Alerts */}
          {activePanel === "alerts" && (
            <div style={{ flex: 1, padding: 14, display: "flex", flexDirection: "column", gap: 10 }}>
              <div style={{ fontSize: 11, fontWeight: 700 }}>Alertes sur {activeSym}</div>
              <div style={{ fontSize: 10, color: text3 }}>Cours : <span style={{ color: text, fontWeight: 700 }}>{fmt(last.c)} FCFA</span></div>
              <button onClick={() => setShowAlertForm(f => !f)}
                style={{ padding: "8px 12px", background: th.blue + "20", border: `1px solid ${th.blue}`, borderRadius: 6, color: th.blue, cursor: "pointer", fontFamily: "inherit", fontSize: 11, fontWeight: 700 }}>
                + Nouvelle alerte
              </button>
              {showAlertForm && (
                <div className="fade-in" style={{ background: bg3, borderRadius: 8, padding: 12, border: `1px solid ${border}`, display: "flex", flexDirection: "column", gap: 8 }}>
                  <div style={{ fontSize: 10, color: text3 }}>Alerter quand le prix atteint :</div>
                  <div style={{ display: "flex", gap: 6 }}>
                    <input value={alertPrice} onChange={e => setAlertPrice(e.target.value)} placeholder={fmt(last.c)}
                      style={{ flex: 1, background: bg2, border: `1px solid ${border}`, borderRadius: 4, padding: "6px 8px", color: text, fontSize: 11, outline: "none" }} />
                    <span style={{ fontSize: 11, color: text3, display: "flex", alignItems: "center" }}>FCFA</span>
                  </div>
                  <div style={{ display: "flex", gap: 6 }}>
                    {[last.c * 0.95, last.c * 1.05, last.c * 1.10].map(p => (
                      <button key={p} onClick={() => setAlertPrice(String(Math.round(p)))}
                        style={{ flex: 1, padding: "4px 0", background: bg2, border: `1px solid ${border}`, borderRadius: 4, color: text3, cursor: "pointer", fontSize: 9, fontFamily: "inherit" }}>
                        {p < last.c ? "▼" : "▲"} {fmtPct((p - last.c) / last.c * 100)}
                      </button>
                    ))}
                  </div>
                  <button onClick={() => { if (alertPrice) { setAlerts(a => [...a, { sym: activeSym, price: +alertPrice, set: last.c }]); setAlertPrice(""); setShowAlertForm(false); }}}
                    style={{ padding: "7px 0", background: th.blue, border: "none", borderRadius: 5, color: "#fff", cursor: "pointer", fontFamily: "inherit", fontSize: 11, fontWeight: 700 }}>
                    Créer
                  </button>
                </div>
              )}
              <div style={{ flex: 1, overflowY: "auto" }}>
                {alerts.length === 0 && <div style={{ color: text3, fontSize: 10, textAlign: "center", marginTop: 20 }}>Aucune alerte</div>}
                {alerts.map((a, i) => {
                  const cur = CANDLE_DATA[a.sym]?.at(-1)?.c || 0;
                  const triggered = (a.price > a.set && cur >= a.price) || (a.price < a.set && cur <= a.price);
                  return (
                    <div key={i} style={{ background: triggered ? th.gold + "15" : bg3, border: `1px solid ${triggered ? th.gold : border}`, borderRadius: 6, padding: 10, marginBottom: 6, display: "flex", justifyContent: "space-between", alignItems: "center" }}>
                      <div>
                        <div style={{ fontSize: 11, fontWeight: 700 }}>{a.sym} → {fmt(a.price)} F</div>
                        <div style={{ fontSize: 9, color: text3 }}>{triggered ? "⚡ DÉCLENCHÉ" : "En attente"}</div>
                      </div>
                      <button onClick={() => setAlerts(al => al.filter((_, j) => j !== i))}
                        style={{ background: "none", border: "none", color: text3, cursor: "pointer", fontSize: 14 }}>×</button>
                    </div>
                  );
                })}
              </div>
            </div>
          )}
        </div>
      </div>

      {/* STATUS BAR */}
      <div style={{ height: 24, background: bg2, borderTop: `1px solid ${border}`, display: "flex", alignItems: "center", paddingInline: 12, gap: 16, flexShrink: 0 }}>
        {[["BRVM-C: 218.45", changePct >= 0 ? th.upC : th.downC], [`BRVM-10: 354.82 ${fmtPct(0.42)}`, th.upC], ["Vol: 2.4M titres", text3], ["Cap: 8 512 Mds FCFA", text3]]
          .map(([label, color], i) => <span key={i} style={{ fontSize: 9, color, fontWeight: 600 }}>{label}</span>)}
        <div style={{ flex: 1 }} />
        <span style={{ fontSize: 9, color: text3 }}>© BONJOUR Finance — Données simulées</span>
        <div className="live-dot" style={{ width: 5, height: 5, borderRadius: "50%", background: th.upC }} />
      </div>
    </div>
  );
}
